# Loop A of debug.md: the reported shape, red 12 runs in 90 (3/30, then 9/60) on a 4-core Linux machine.
#
# `socket_state_follows_the_trade_socket` as it is on main at 5c63bd5, without its wait for the
# first `subscribe` before the drop. Append it to tests/exec_client.rs in a throwaway worktree,
# build with `cargo test --test exec_client --no-run`, then run the WHOLE binary many times: run
# alone, the test went red 0 times in 100, and 1 in 40 under CPU load, because the race needs the
# other tests competing for the cores.
#
#   for i in $(seq 1 90); do target/debug/deps/exec_client-<hash> || echo "run $i red"; done
#
# A red run ends with `venue refused: ["subscribe before auth"]` from `Harness::drop`.

#[tokio::test(flavor = "multi_thread")]
async fn drop_right_after_the_first_connected() {
    let (sender, mut system) = tokio::sync::mpsc::unbounded_channel();
    replace_system_event_sender(sender);
    let registry = SocketReconnectRegistry::default();
    let (mut h, connected) = Harness::start_in(&registry, |_, _| {}).await;
    connected.unwrap();
    assert_eq!(next_state(&mut system).await, SocketState::Connected);
    h.venue.drop_socket();
    assert_eq!(next_state(&mut system).await, SocketState::Disconnected);
    assert_eq!(next_state(&mut system).await, SocketState::Connected);
    h.venue.wait_frames("auth", 2).await;
    h.client.disconnect().await.unwrap();
    tokio::time::sleep(Duration::from_millis(500)).await;
    assert!(system.try_recv().is_err(), "disconnect reported a state");
    h.client.connect().await.unwrap();
    assert_eq!(next_state(&mut system).await, SocketState::Connected);
    h.client.disconnect().await.unwrap();
}
